# ============================================================================
# USUARIOS_SERV - Servicio de Gestión de Usuarios, Perfiles y Preferencias
# ============================================================================
# Este servicio maneja toda la información de usuarios del sistema
# Puerto: 8000


# ============================================================================
# AUTH_SERV - Servicio de Autenticación y Validación de Tokens JWT
# ============================================================================
# Este servicio maneja login, tokens JWT y validación de credenciales
# Puerto: 8001
# Endpoints: POST /api/auth/login-token/, POST /api/auth/validate-token/


# ============================================================================
# MASCOTAS_SERV - Servicio de Reportes y Contactos de Mascotas
# ============================================================================
# Este servicio maneja reportes de mascotas perdidas/encontradas y contactos
# Puerto: 8002
# Endpoints: /api/reportes/, /api/contactos/
# Requiere validación de tokens JWT y role Admin para modificaciones


# BFF_SERV - Backend For Frontend (Capa Proxy/Agregador)
# ============================================================================
# Este servicio es el punto de entrada único para el frontend
# Proxifica solicitudes a todos los microservicios backend
# Puerto: 8003
# Servicios que proxifica:
#   - auth_serv (puerto 8001) - login y refresh
#   - usuarios_serv (puerto 8000) - usuarios, perfiles, preferencias
#   - mascotas_serv (puerto 8002) - reportes y contactos


# ============================================================================
# NOTIFICACIONES_SERV - Servicio de Notificaciones Asincrónicas
# ============================================================================
# Este servicio maneja notificaciones por correo, SMS y push
# Utiliza Celery para tareas asincrónicas + Redis como broker
# Puerto: 8004
